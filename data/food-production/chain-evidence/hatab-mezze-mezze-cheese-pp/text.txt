updateHeaderHeight() {

const height = Math.round(header.getBoundingClientRect().height);
this.headerHeight = height;
if (Alpine.store('header')) {
Alpine.store('header').headerHeight = height;
}
document.documentElement.style.setProperty('--header-height', `${height}px`);// For sticky elements: if safe area exists (PWA), use full height, otherwise subtract notification/topbar
const safeAreaTop = parseInt(getComputedStyle(document.documentElement).getPropertyValue('--safe-area-inset-top')) || 0;
const stickyHeaderHeight = safeAreaTop > 0 ? height : (height - this.notificationHeight - this.topBarHeight);
document.documentElement.style.setProperty('--sticky-header-height', `${stickyHeaderHeight}px`);
} else {
// Header is hidden, set heights to 0
this.headerHeight = 0;
if (Alpine.store('header')) {
Alpine.store('header').headerHeight = 0;
}
document.documentElement.style.setProperty('--header-height', '0px');
document.documentElement.style.setProperty('--sticky-header-height', '0px');
}
